# Referral Rewards And Vouchers Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an onboarding referral scheme with clear signup-wizard code entry, accepted-referral tracking at wizard completion, and one-time in-app voucher issuance/redemption for both inviter and invitee.

**Architecture:** Model the feature as a dedicated `rewards` domain in Supabase with separate ledgers for codes, referrals, and vouchers, then expose only RPC-shaped mutations to the Flutter app. The app keeps referral entry in onboarding, moves reward inventory into a dedicated profile flow, and removes the current post-signup modal prompt so there is one canonical referral path.

**Tech Stack:** Flutter, Provider, Supabase RPCs, Postgres schema migrations, RLS, `share_plus`, existing `SupabaseService` helper pattern

Write the new migration so it creates the domain tables, constraints, indexes, and comments:

```sql
create schema if not exists rewards;

create table rewards.referral_codes (
  id uuid primary key default gen_random_uuid(),
  owner_user_id uuid not null references public.users (supabase_id) on delete cascade,
  code text not null,
  normalized_code text generated always as (upper(trim(code))) stored,
  is_active boolean not null default true,
  created_at timestamptz not null default timezone('utc', now()),
  updated_at timestamptz not null default timezone('utc', now()),
  unique (owner_user_id),
  unique (normalized_code)
);

create table rewards.referrals (
  id uuid primary key default gen_random_uuid(),
  referral_code_id uuid not null references rewards.referral_codes (id) on delete restrict,
  inviter_user_id uuid not null references public.users (supabase_id) on delete cascade,
  invitee_user_id uuid not null references public.users (supabase_id) on delete cascade,
  entered_code text not null,
  status text not null check (status in ('pending', 'accepted', 'voided')),
  entered_at timestamptz not null default timezone('utc', now()),
  accepted_at timestamptz,
  voided_at timestamptz,
  acceptance_trigger text,
  metadata jsonb not null default '{}'::jsonb,
  unique (invitee_user_id),
  check (inviter_user_id <> invitee_user_id)
);

create table rewards.vouchers (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references public.users (supabase_id) on delete cascade,
  source_referral_id uuid references rewards.referrals (id) on delete set null,
  source_type text not null check (source_type in ('inviter_reward', 'invitee_reward')),
  campaign_key text not null default 'imperial_farmers_market_10pct_selected_stores',
  title text not null,
  description text not null,
  discount_percent integer not null check (discount_percent = 10),
  merchant_name text not null default 'Imperial Farmers Market',
  terms_text text not null,
  status text not null check (status in ('available', 'redeemed', 'expired', 'voided')),
  issued_at timestamptz not null default timezone('utc', now()),
  redeemed_at timestamptz,
  expires_at timestamptz,
  redemption_token uuid not null default gen_random_uuid(),
  metadata jsonb not null default '{}'::jsonb
);

create unique index rewards_vouchers_referral_user_source_unique
  on rewards.vouchers (source_referral_id, user_id, source_type)
  where source_referral_id is not null;
```

  return v_row;
end;
$$;

- [ ] **Step 2: Implement `rewards.apply_referral_code(p_code text)`**

The function should validate the code, block self-referral, and upsert one pending referral for the authenticated invitee:

```sql
create or replace function rewards.apply_referral_code(p_code text)
returns rewards.referrals
language plpgsql
security definer
set search_path = rewards, public
as $$
declare
  v_code rewards.referral_codes;
  v_referral rewards.referrals;
begin
  if auth.uid() is null then
    raise exception 'Authentication required';
  end if;

  select *
  into v_code
  from rewards.referral_codes
  where normalized_code = upper(trim(p_code))
    and is_active = true;

  if not found then
    raise exception 'Invalid referral code';
  end if;

  if v_code.owner_user_id = auth.uid() then
    raise exception 'You cannot use your own referral code';
  end if;

  insert into rewards.referrals (
    referral_code_id,
    inviter_user_id,
    invitee_user_id,
    entered_code,
    status
  )
  values (
    v_code.id,
    v_code.owner_user_id,
    auth.uid(),
    upper(trim(p_code)),
    'pending'
  )
  on conflict (invitee_user_id) do update
    set referral_code_id = excluded.referral_code_id,
        inviter_user_id = excluded.inviter_user_id,
        entered_code = excluded.entered_code,
        status = case
          when rewards.referrals.status = 'accepted' then rewards.referrals.status
          else 'pending'
        end
  returning * into v_referral;

  if v_referral.status = 'accepted' then
    raise exception 'Referral code is already locked for this account';
  end if;

  return v_referral;
end;
$$;
```

- [ ] **Step 3: Implement `rewards.accept_pending_referral_for_user(p_user_id uuid)`**

This function must be idempotent and issue exactly one voucher to each side:

```sql
create or replace function rewards.accept_pending_referral_for_user(p_user_id uuid)
returns void
language plpgsql
security definer
set search_path = rewards, public
as $$
declare
  v_referral rewards.referrals;
begin
  select *
  into v_referral
  from rewards.referrals
  where invitee_user_id = p_user_id
    and status = 'pending'
  for update;

  if not found then
    return;
  end if;

  update rewards.referrals
  set status = 'accepted',
      accepted_at = timezone('utc', now()),
      acceptance_trigger = 'wizard_completion'
  where id = v_referral.id;

  insert into rewards.vouchers (
    user_id, source_referral_id, source_type, title, description, terms_text, status
  )
  values
    (
      v_referral.invitee_user_id,
      v_referral.id,
      'invitee_reward',
      '10% off Imperial Farmers Market at Selected Stores',
      'One-time use reward unlocked after completing signup with a referral code.',
      'Valid for one use at selected Imperial Farmers Market stores.',
      'available'
    ),
    (
      v_referral.inviter_user_id,
      v_referral.id,
      'inviter_reward',
      '10% off Imperial Farmers Market at Selected Stores',
      'One-time use reward unlocked when your referral completed signup.',
      'Valid for one use at selected Imperial Farmers Market stores.',
      'available'
    )
  on conflict (source_referral_id, user_id, source_type)
    where source_referral_id is not null
    do nothing;
end;
$$;
```

- [ ] **Step 4: Implement dashboard and redemption RPCs**

Keep the app read shape simple by returning one aggregated payload and one atomic redeem mutation:

```sql
create or replace function rewards.get_my_referral_dashboard()
returns jsonb
language sql
security definer
set search_path = rewards, public
as $$
  select jsonb_build_object(
    'referral_code', (select code from rewards.ensure_referral_code()),
    'accepted_referral_count',
      (select count(*) from rewards.referrals where inviter_user_id = auth.uid() and status = 'accepted'),
    'available_vouchers',
      coalesce(
        (select jsonb_agg(to_jsonb(v) order by v.issued_at desc) from rewards.vouchers v where v.user_id = auth.uid() and v.status = 'available'),
        '[]'::jsonb
      ),
    'used_vouchers',
      coalesce(
        (select jsonb_agg(to_jsonb(v) order by v.redeemed_at desc nulls last, v.issued_at desc) from rewards.vouchers v where v.user_id = auth.uid() and v.status = 'redeemed'),
        '[]'::jsonb
      )
  );
$$;

create or replace function rewards.redeem_voucher(p_voucher_id uuid)
returns rewards.vouchers
language plpgsql
security definer
set search_path = rewards, public
as $$
declare
  v_voucher rewards.vouchers;
begin
  update rewards.vouchers
  set status = 'redeemed',
      redeemed_at = timezone('utc', now())
  where id = p_voucher_id
    and user_id = auth.uid()
    and status = 'available'
  returning * into v_voucher;

  if not found then
    raise exception 'Voucher is not available';
  end if;

  return v_voucher;
end;
$$;
```

class ReferralDashboard {
  final String referralCode;
  final int acceptedReferralCount;
  final List<RewardVoucher> availableVouchers;
  final List<RewardVoucher> usedVouchers;

  bool get hasAnyVouchers =>
      availableVouchers.isNotEmpty || usedVouchers.isNotEmpty;

  factory ReferralDashboard.fromJson(Map<String, dynamic> json) {
    List<RewardVoucher> parseList(String key) {
      final raw = (json[key] as List?) ?? const [];
      return raw
          .whereType<Map>()
          .map((item) => RewardVoucher.fromJson(Map<String, dynamic>.from(item)))
          .toList();

    return ReferralDashboard(
      referralCode: json['referral_code'] as String? ?? '',
      acceptedReferralCount:
          (json['accepted_referral_count'] as num?)?.toInt() ?? 0,
      availableVouchers: parseList('available_vouchers'),

class RewardsHelper {
  RewardsHelper({SupabaseClient? client})
      : _client = client ?? SupabaseClientManager().client;

  final SupabaseClient _client;

  Future<void> applyReferralCode(String code) async {
    await _client
        .schema('rewards')
        .rpc('apply_referral_code', params: {'p_code': code});
  }

  Future<ReferralDashboard> getMyDashboard() async {
    final response = await _client.schema('rewards').rpc(
          'get_my_referral_dashboard',
        );
    return ReferralDashboard.fromJson(Map<String, dynamic>.from(response));
  }

  Future<RewardVoucher> redeemVoucher(String voucherId) async {
    final response = await _client.schema('rewards').rpc(
      'redeem_voucher',
      params: {'p_voucher_id': voucherId},
    );
    return RewardVoucher.fromJson(Map<String, dynamic>.from(response));
  }
}
